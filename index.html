◈ morven.js

MorvenWallet Auth SDK — CDN edition

v1.0.0

Quick Start

<script src="https://morven-sdk.vercel.app/morven.js"></script>
<script>
  // Login
  const { jwt, user } = await morvenWallet.login();

  // Get balance
  const { mrv } = await morvenWallet.getBalance(jwt);

  // Pay with MRV
  await morvenWallet.pay(jwt, 10, 'letterforge-pro');

  // Render button
  morvenWallet.renderButton('#login-area', {
    onLogin: ({ user }) => console.log('Welcome', user.display_name)
  });
</script>

CDN URL

https://morven-sdk.vercel.app/morven.js

Methods

morvenWallet.login()                       → { jwt, user }
morvenWallet.getBalance(jwt)               → { mrv, usdt, usd_value }
morvenWallet.pay(jwt, amount, productId)   → { success, tx_id, new_balance }
morvenWallet.airdrop(walletId, amt, reason, key) → { success, tx_id }
morvenWallet.logout()
morvenWallet.isLoggedIn()                  → boolean
morvenWallet.renderButton(selector, opts)
